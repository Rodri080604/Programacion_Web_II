GO -- Procedimiento para validar login CREATE PROCEDURE ValidarUsuario @Username VARCHAR(50), @Password VARCHAR(255) AS BEGIN SELECT CI_Personal, Rol, Nombres + ' ' + Apellido_1 AS Nombre FROM PERSONAL WHERE Username = @Username AND Password = @Password AND Estado = 'Activo'; END; GO -- Procedimiento para actualizar horarios ajustados CREATE PROCEDURE ActualizarHorariosAjustados @NombreHorario VARCHAR(50) AS BEGIN -- Desactivar horarios actuales UPDATE HORARIOS_AJUSTADOS SET ES_ACTIVO = 0 WHERE ES_ACTIVO = 1; -- Eliminar filas existentes para este tipo de horario DELETE FROM HORARIOS_AJUSTADOS WHERE TIPO_HORARIO = @NombreHorario; -- Obtener minutos adicionales DECLARE @MinutosAdicionales INT; SELECT @MinutosAdicionales = MINUTOS_ADICIONALES FROM TIPO_HORARIO WHERE NOMBRE = @NombreHorario; -- Turno Mañana: Ajustar entrada, mantener salida fija en 12:20 DECLARE @InicioPrimerPeriodoManana TIME = DATEADD(MINUTE, @MinutosAdicionales, '07:00:00'); DECLARE @FinUltimoPeriodoManana TIME = '12:20:00'; -- Salida fija DECLARE @TiempoTotalManana INT = DATEDIFF(MINUTE, @InicioPrimerPeriodoManana, @FinUltimoPeriodoManana); DECLARE @TiempoRecesosManana INT = 20; -- 2 recesos de 10 minutos DECLARE @TiempoClasesManana INT = @TiempoTotalManana - @TiempoRecesosManana; DECLARE @DuracionPeriodoManana INT = @TiempoClasesManana / 6; -- Turno Tarde: Mantener entrada fija en 14:00, ajustar salida DECLARE @InicioPrimerPeriodoTarde TIME = '14:00:00'; -- Entrada fija DECLARE @FinUltimoPeriodoTarde TIME = DATEADD(MINUTE, @MinutosAdicionales, '19:20:00'); -- Salida ajustada DECLARE @TiempoTotalTarde INT = DATEDIFF(MINUTE, @InicioPrimerPeriodoTarde, @FinUltimoPeriodoTarde); DECLARE @TiempoRecesosTarde INT = 20; -- 2 recesos de 10 minutos DECLARE @TiempoClasesTarde INT = @TiempoTotalTarde - @TiempoRecesosTarde; DECLARE @DuracionPeriodoTarde INT = @TiempoClasesTarde / 6; -- CTE recursiva para calcular horarios de mañana WITH HorariosManana AS ( SELECT 1 AS Orden, CAST(@InicioPrimerPeriodoManana AS TIME) AS HoraInicio, CAST(DATEADD(MINUTE, @DuracionPeriodoManana, @InicioPrimerPeriodoManana) AS TIME) AS HoraFin UNION ALL SELECT H.Orden + 1, CAST(CASE WHEN H.Orden = 2 OR H.Orden = 4 THEN DATEADD(MINUTE, 10, H.HoraFin) ELSE H.HoraFin END AS TIME), CAST(DATEADD(MINUTE, @DuracionPeriodoManana, CASE WHEN H.Orden = 2 OR H.Orden = 4 THEN DATEADD(MINUTE, 10, H.HoraFin) ELSE H.HoraFin END) AS TIME) FROM HorariosManana H WHERE H.Orden < 6 ), -- CTE recursiva para calcular horarios de tarde HorariosTarde AS ( SELECT 7 AS Orden, CAST(@InicioPrimerPeriodoTarde AS TIME) AS HoraInicio, CAST(DATEADD(MINUTE, @DuracionPeriodoTarde, @InicioPrimerPeriodoTarde) AS TIME) AS HoraFin UNION ALL SELECT H.Orden + 1, CAST(CASE WHEN H.Orden = 8 OR H.Orden = 10 THEN DATEADD(MINUTE, 10, H.HoraFin) ELSE H.HoraFin END AS TIME), CAST(DATEADD(MINUTE, @DuracionPeriodoTarde, CASE WHEN H.Orden = 8 OR H.Orden = 10 THEN DATEADD(MINUTE, 10, H.HoraFin) ELSE H.HoraFin END) AS TIME) FROM HorariosTarde H WHERE H.Orden < 12 ) -- Combinar mañana y tarde INSERT INTO HORARIOS_AJUSTADOS (ID_PERIODO, ORDEN, TURNO, HORA_INICIO_AJUSTADA, HORA_FIN_AJUSTADA, TIPO_HORARIO, MINUTOS_ADICIONALES, ES_ACTIVO) SELECT P.ID_PERIODO, H.Orden, P.TURNO, H.HoraInicio, H.HoraFin, @NombreHorario, @MinutosAdicionales, 1 FROM ( SELECT Orden, HoraInicio, HoraFin FROM HorariosManana UNION ALL SELECT Orden, HoraInicio, HoraFin FROM HorariosTarde ) H JOIN PERIODOS P ON H.Orden = P.ORDEN; -- Asegurar que el último periodo de la mañana termine en 12:20 UPDATE HORARIOS_AJUSTADOS SET HORA_FIN_AJUSTADA = '12:20:00' WHERE TIPO_HORARIO = @NombreHorario AND ORDEN = 6 AND ES_ACTIVO = 1; -- Asegurar que el primer periodo de la tarde comience en 14:00 UPDATE HORARIOS_AJUSTADOS SET HORA_INICIO_AJUSTADA = '14:00:00' WHERE TIPO_HORARIO = @NombreHorario AND ORDEN = 7 AND ES_ACTIVO = 1; END; GO -- Procedimiento para agregar un nuevo horario CREATE PROCEDURE AgregarNuevoHorario @NombreHorario VARCHAR(50), @MinutosAdicionales INT, @TiempoTolerancia INT AS BEGIN IF NOT EXISTS (SELECT 1 FROM TIPO_HORARIO WHERE NOMBRE = @NombreHorario) BEGIN INSERT INTO TIPO_HORARIO (NOMBRE, MINUTOS_ADICIONALES, TIEMPO_TOLERANCIA) VALUES (@NombreHorario, @MinutosAdicionales, @TiempoTolerancia); END EXEC ActualizarHorariosAjustados @NombreHorario; END; GO -- Procedimiento para imprimir el horario de una persona CREATE PROCEDURE ImprimirHorarioUsuario @CIPersonal VARCHAR(20) AS BEGIN SELECT P.Nombres + ' ' + P.Apellido_1 AS NombreUsuario, HP.DIA_SEMANA AS Dia, HA.TURNO AS Turno, MIN(HA.HORA_INICIO_AJUSTADA) AS HoraEntrada, MAX(HA.HORA_FIN_AJUSTADA) AS HoraSalida FROM PERSONAL P JOIN HORARIO_PERSONAL HP ON P.CI_Personal = HP.CI_Personal JOIN HORARIOS_AJUSTADOS HA ON HP.ID_PERIODO = HA.ID_PERIODO WHERE P.CI_Personal = @CIPersonal AND HA.ES_ACTIVO = 1 GROUP BY P.Nombres, P.Apellido_1, HP.DIA_SEMANA, HA.TURNO ORDER BY CASE HP.DIA_SEMANA WHEN 'Lunes' THEN 1 WHEN 'Martes' THEN 2 WHEN 'Miercoles' THEN 3 WHEN 'Jueves' THEN 4 WHEN 'Viernes' THEN 5 ELSE 6 END, HA.TURNO; END; GO -- Procedimiento: Mostrar personal con foto de perfil CREATE PROCEDURE MostrarPersonalConFoto AS BEGIN SELECT CI_Personal, Nombres + ' ' + Apellido_1 AS Nombre, Foto_Perfil FROM PERSONAL WHERE Estado = 'Activo' ORDER BY Nombres; END; GO -- Procedimiento: Lista de profesores con salidas anticipadas CREATE PROCEDURE ReporteSalidasAnticipadas @FechaInicio DATE, @FechaFin DATE AS BEGIN SELECT P.Nombres + ' ' + P.Apellido_1 AS Nombre, COUNT(A.ID_Asistencia) AS NumeroSalidasAnticipadas FROM PERSONAL P JOIN ASIGNACION_CARGOS AC ON P.CI_Personal = AC.CI_Personal JOIN CARGOS C ON AC.ID_Cargo = C.ID_Cargo JOIN ASISTENCIA A ON P.CI_Personal = A.CI_Personal WHERE C.Tipo_Cargo = 'Docente' AND A.Estado = 'Salida Anticipada' AND A.Fecha BETWEEN @FechaInicio AND @FechaFin AND AC.Estado = 'Activo' GROUP BY P.Nombres, P.Apellido_1 ORDER BY NumeroSalidasAnticipadas DESC; END; GO -- Procedimiento: Personal con faltas o retrasos en una fecha CREATE PROCEDURE PersonalFaltasRetrasosFecha @Fecha DATE AS BEGIN SELECT P.Nombres + ' ' + P.Apellido_1 AS Nombre, A.Estado, A.Minutos_Retraso FROM PERSONAL P JOIN ASISTENCIA A ON P.CI_Personal = A.CI_Personal WHERE A.Fecha = @Fecha AND A.Estado IN ('Falta', 'Retraso') ORDER BY P.Nombres; END; GO -- Procedimiento: Minutos acumulados por retraso por tipo de personal CREATE PROCEDURE ReporteMinutosRetraso @FechaInicio DATE, @FechaFin DATE, @TipoCargo VARCHAR(20) AS BEGIN SELECT P.Nombres + ' ' + P.Apellido_1 AS Nombre, SUM(A.Minutos_Retraso) AS MinutosRetraso FROM PERSONAL P JOIN ASIGNACION_CARGOS AC ON P.CI_Personal = AC.CI_Personal JOIN CARGOS C ON AC.ID_Cargo = C.ID_Cargo JOIN ASISTENCIA A ON P.CI_Personal = A.CI_Personal WHERE A.Fecha BETWEEN @FechaInicio AND @FechaFin AND A.Estado = 'Retraso' AND C.Tipo_Cargo = @TipoCargo AND AC.Estado = 'Activo' GROUP BY P.Nombres, P.Apellido_1 ORDER BY MinutosRetraso DESC; END; GO -- Procedimiento: Reporte de faltas, retrasos y salidas anticipadas CREATE PROCEDURE ReporteFaltasRetrasosAnticipadas @FechaInicio DATE, @FechaFin DATE AS BEGIN SELECT P.Nombres + ' ' + P.Apellido_1 AS Nombre, SUM(CASE WHEN A.Estado = 'Falta' THEN 1 ELSE 0 END) AS Faltas, SUM(CASE WHEN A.Estado = 'Retraso' THEN 1 ELSE 0 END) AS Retrasos, SUM(CASE WHEN A.Estado = 'Salida Anticipada' THEN 1 ELSE 0 END) AS SalidasAnticipadas FROM PERSONAL P JOIN ASISTENCIA A ON P.CI_Personal = A.CI_Personal WHERE A.Fecha BETWEEN @FechaInicio AND @FechaFin GROUP BY P.Nombres, P.Apellido_1 ORDER BY P.Nombres; END; GO -- Procedimiento: Personal con licencias a aprobar CREATE PROCEDURE ListaLicenciasPendientes AS BEGIN SELECT P.Nombres + ' ' + P.Apellido_1 AS Nombre, COUNT(L.ID_Licencia) AS NumeroLicencias FROM PERSONAL P JOIN LICENCIA L ON P.CI_Personal = L.CI_Personal WHERE L.Estado = 'Pendiente' GROUP BY P.Nombres, P.Apellido_1 ORDER BY NumeroLicencias DESC; END; GO -- Procedimiento: Lista de personal femenino CREATE PROCEDURE ListaMujeres AS BEGIN SELECT Nombres + ' ' + Apellido_1 AS Nombre FROM PERSONAL WHERE Genero = 'F' ORDER BY Nombres; END; GO -- Procedimiento: Lista de padres de familia CREATE PROCEDURE ListaPadres AS BEGIN SELECT Nombres + ' ' + Apellido_1 AS Nombre, Num_Hijos FROM PERSONAL WHERE Num_Hijos > 0 ORDER BY Nombres; END; GO -- Procedimiento: Rango de fechas de horarios por gestión CREATE PROCEDURE RangoFechasHorarios @Gestion VARCHAR(10) AS BEGIN SELECT TH.NOMBRE AS TipoHorario, MIN(GH.Fecha_Inicio) AS FechaInicio, MAX(GH.Fecha_Fin) AS FechaFin FROM GESTION_HORARIO GH JOIN TIPO_HORARIO TH ON GH.ID_Tipo_Horario = TH.ID_TIPO_HORARIO WHERE GH.Gestion = @Gestion GROUP BY TH.NOMBRE ORDER BY FechaInicio; END; GO -- Procedimiento: Historial de cargos de una persona CREATE PROCEDURE HistorialCargosPersona @CIPersonal VARCHAR(20) AS BEGIN SELECT C.Nombre_Cargo, C.Tipo_Cargo, AC.Fecha_Inicio, AC.Fecha_Fin FROM ASIGNACION_CARGOS AC JOIN CARGOS C ON AC.ID_Cargo = C.ID_Cargo JOIN PERSONAL P ON AC.CI_Personal = P.CI_Personal WHERE P.CI_Personal = @CIPersonal ORDER BY AC.Fecha_Inicio; END; GO -- Procedimiento: Planilla de cargos CREATE PROCEDURE PlanillaCargos AS BEGIN SELECT P.Nombres + ' ' + P.Apellido_1 AS Nombre, C.Nombre_Cargo, C.Tipo_Cargo FROM PERSONAL P JOIN ASIGNACION_CARGOS AC ON P.CI_Personal = AC.CI_Personal JOIN CARGOS C ON AC.ID_Cargo = C.ID_Cargo WHERE AC.Estado = 'Activo' ORDER BY P.Nombres; END; GO -- Procedimiento: Reporte de atrasos CREATE PROCEDURE ReporteAtrasos @FechaInicio DATE, @FechaFin DATE AS BEGIN SELECT P.Nombres + ' ' + P.Apellido_1 AS Nombre, SUM(A.Minutos_Retraso) AS MinutosRetraso FROM PERSONAL P JOIN ASISTENCIA A ON P.CI_Personal = A.CI_Personal WHERE A.Fecha BETWEEN @FechaInicio AND @FechaFin AND A.Estado = 'Retraso' GROUP BY P.Nombres, P.Apellido_1 ORDER BY MinutosRetraso DESC; END; GO -- Procedimiento: Días no laborales CREATE PROCEDURE ReporteDiasNoLaborales @FechaInicio DATE, @FechaFin DATE AS BEGIN SELECT Fecha, Tipo_Feriado FROM FERIADO WHERE Fecha BETWEEN @FechaInicio AND @FechaFin ORDER BY Fecha;
