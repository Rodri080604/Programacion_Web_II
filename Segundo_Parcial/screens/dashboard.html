Películas: Contiene información básica de las películas, como título, duración, género y clasificación. El campo id_pelicula es la clave primaria. Salas: Registra las salas del cine con un nombre y su capacidad de asientos. El campo id_sala es la clave primaria. Horarios: Vincula películas y salas con una fecha y hora específica de proyección. Usa claves foráneas para relacionar con peliculas y salas. Clientes: Almacena datos de los clientes, con un campo email único para evitar duplicados. El campo id_cliente es la clave primaria. Boletos: Registra cada boleto vendido, vinculando al cliente, el horario y el asiento asignado. Incluye el precio y la fecha de compra. -- Creación de la base de datos CREATE DATABASE cine_admin; USE cine_admin; -- Tabla para almacenar las películas CREATE TABLE peliculas ( id_pelicula INT AUTO_INCREMENT PRIMARY KEY, titulo VARCHAR(100) NOT NULL, duracion INT NOT NULL, -- Duración en minutos genero VARCHAR(50), clasificacion VARCHAR(10) -- Ejemplo: PG, PG-13, R ); -- Tabla para almacenar las salas CREATE TABLE salas ( id_sala INT AUTO_INCREMENT PRIMARY KEY, nombre VARCHAR(50) NOT NULL, capacidad INT NOT NULL -- Número de asientos ); -- Tabla para almacenar los horarios de proyección CREATE TABLE horarios ( id_horario INT AUTO_INCREMENT PRIMARY KEY, id_pelicula INT, id_sala INT, fecha_hora DATETIME NOT NULL, FOREIGN KEY (id_pelicula) REFERENCES peliculas(id_pelicula), FOREIGN KEY (id_sala) REFERENCES salas(id_sala) ); -- Tabla para almacenar los clientes CREATE TABLE clientes ( id_cliente INT AUTO_INCREMENT PRIMARY KEY, nombre VARCHAR(100) NOT NULL, email VARCHAR(100) UNIQUE, telefono VARCHAR(15) ); -- Tabla para almacenar los boletos CREATE TABLE boletos ( id_boleto INT AUTO_INCREMENT PRIMARY KEY, id_cliente INT, id_horario INT, asiento VARCHAR(10), -- Ejemplo: A1, B12 precio DECIMAL(10, 2) NOT NULL, fecha_compra DATETIME NOT NULL, FOREIGN KEY (id_cliente) REFERENCES clientes(id_cliente), FOREIGN KEY (id_horario) REFERENCES horarios(id_horario) );
